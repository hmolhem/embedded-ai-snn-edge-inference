\newtcolorbox{proposalbox}[1]{
    colback=blue!3,
    colframe=blue!55!black,
    title={#1},
    fonttitle=\bfseries,
    arc=2mm,
    boxrule=0.5pt
}

\newtcolorbox{resultbox}[1]{
    colback=green!3,
    colframe=green!45!black,
    title={#1},
    fonttitle=\bfseries,
    arc=2mm,
    boxrule=0.5pt
}

\newtcolorbox{riskbox}[1]{
    colback=orange!4,
    colframe=orange!70!black,
    title={#1},
    fonttitle=\bfseries,
    arc=2mm,
    boxrule=0.5pt
}
